int CAR-FLEET(int target, int[] position, int[] speed)
    Stack<Double> stack = new Stack<>();
    int[][] combine = new int[position.length][2];
    for (int i = 0; i < position.length; i++)
    	combine[i][0] = position[i]; combine[i][1] = speed[i];
    Arrays.sort(combine, java.util.Comparator.comparingInt(o -> o[0]));
    for (int i = combine.length - 1; i >= 0; i--)
    	double currentTime = (double) (target - combine[i][0]) / combine[i][1];
    	if (!stack.isEmpty() && currentTime <= stack.peek())
    		continue;
    	else
    		stack.push(currentTime);
    return stack.size();

String REMOVE-K-DIGITS(String nums, int k)
        Stack<Character> st = new Stack<>();
        for (int i = 0; i < nums.length(); i++)
            char digit = nums.charAt(i); // Current digit
            while (!st.isEmpty() && k > 0 && st.peek() > digit)
                st.pop(); // Remove the last digit
                k--; // Decrement k by 1
            st.push(digit);
        while (k > 0)
            st.pop(); // Pop the last added digits
            k--; // Decrement k by 1
        if (st.isEmpty()) return "0";
        StringBuilder res = new StringBuilder();
        while (!st.isEmpty()) res.append(st.pop());
        while (res.length() > 0 && res.charAt(res.length() - 1) == '0')
            res.deleteCharAt(res.length() - 1);
        res.reverse();
        if (res.length() == 0) return "0";
        return res.toString();

int BASIC-CALCULATOR(String s)

    Stack<Integer> stack = new Stack<Integer>();
    for (int i = 0; i < len; i++)
    	if (Character.isDigit(s.charAt(i)))
    		int sum = s.charAt(i) - '0';
    		while (i + 1 < len && Character.isDigit(s.charAt(i + 1)))
    			sum = sum * 10 + s.charAt(i + 1) - '0';
    			i++;
    		result += sum * sign;
    	 else if (s.charAt(i) == '+') sign = 1;
    	 else if (s.charAt(i) == '-') sign = -1;
    	 else if (s.charAt(i) == '(')
    		stack.push(result); stack.push(sign);
    		result = 0; sign = 1;
    	 else if (s.charAt(i) == ')')
    		result = result * stack.pop() + stack.pop();
    return result;

int CELEBRITY-PROBLEM(int[][] M)
    int n = M.length;
    int top = 0, down = n - 1;
    while (top < down) {
        if (M[top][down] == 1) top = top + 1;
        else if (M[down][top] == 1) down = down - 1;
        else top++; down--;
    if (top > down) return -1;
    for (int i = 0; i < n; i++)
        if (i == top) continue;
        if (M[top][i] == 1 || M[i][top] == 0)
            return -1;
    return top;

// Function to find indices of Next Smaller Element (NSE)
int[] findNSE(int[] arr) {
    int n = arr.length;
    int[] ans = new int[n];
    Stack<Integer> st = new Stack<>();
    for (int i = n - 1; i >= 0; i--)
        while (!st.isEmpty() && arr[st.peek()] >= arr[i])
            st.pop();
        ans[i] = !st.isEmpty() ? st.peek() : n;
        st.push(i);
    return ans;
