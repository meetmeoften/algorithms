void PERMUTATIONS(int[] nums, List<List<Integer>> result, List<Integer> temp) 
    if(temp.size() == nums.length)
        result.add(new ArrayList<>(temp));
        return;
    for(int i= 0; i< nums.length; i++)
        if(temp.contains(nums[i]))
            continue;
        temp.add(nums[i]);
        helper(nums, result, temp);
        temp.remove(temp.size()-1);


void SUBSETS(int[] nums, int idx, List<List<Integer>> result, List<Integer> temp)
    result.add(new ArrayList<>(temp));
    for(int i= idx; i< nums.length; i++)

void PHONE-NUMBER-COMBINATION(String digits, int idx, StringBuilder sb, List<String> result,
                Map<Character, String> map)
    if (sb.length() == digits.length())
    	result.add(sb.toString());
    	return;
    String letters = map.get(digits.charAt(idx));
    for (int i = 0; i < letters.length(); i++)
    	sb.append(letters.charAt(i));
    	helper(digits, idx + 1, sb, result, map);
    	sb.deleteCharAt(sb.length() - 1);
