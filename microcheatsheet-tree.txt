Node INVERT-TREE(Node root)

boolean VALIDATE-BST(Node root, Integer min, Integer max)
    if(root.val < min || root.val >= max)
        return false;
    if(root.left != null && !helper(root.left, min, root.val))
        return false;
    if(root.right != null && !helper(root.right, root.val, max))
        return false;
    return true;

Node LCA(Node root, Integer node1, Integer node2)
    if (root == null) return null;
    if (root.data == node1 || root.data == node2)
        return root;
    Node left = lca(root.left, node1, node2);
    Node right = lca(root.right, node1, node2);
    if (left == null) return right;
    else if (right == null) return left;
    else return root;

boolean IS-CHILDREN-SUM(Node root)
    if(root == null || (root.left == null & root.right == null)
        return true
    int leftVal = (root.left == null) ? 0 : root.left.val;
    int rightVal = (root.right == null) ? 0 : root.right.val;
    return (root.val == leftVal + rightVal) &&
            isChildrenSum(root.left) && isChildrenSum(root.right)

void CONVERT-TO-CHILDREN-SUM(Node root)
    if(root == null || (root.left == null & root.right == null)
        return true
    convertToChildrenSum(root.left)
    convertToChildrenSum(root.right)
    int leftVal = (root.left == null) ? 0 : root.left.val;
    int rightVal = (root.right == null) ? 0 : root.right.val;
    int childSum = leftVal + rightVal
    if(childSum >= root.val)
        root.val = childSum
    else if(root.left != null)
        root.left.val = root.val - childSum
        convertToChildrenSum(root.left)
    else if(root.right != null)
        root.right.val = root.val - childSum
        convertToChildrenSum(root.right)

int DIAMETER-HEIGHT(Node node)
    if (node == null) return 0;
    int leftHeight = height(node.left);
    int rightHeight = height(node.right);
    diameter = Math.max(diameter, leftHeight + rightHeight);
    return 1 + Math.max(leftHeight, rightHeight);

boolean IDENTICAL(Node node1, Node node2)
    if (node1 == null && node2 == null) return true;
    return node1 != null && node2 != null &&
            node1.data == node2.data &&
            identical(node1.left, node2.left) &&
            identical(node1.right, node2.right);

int MIN-TIME-TO-BURN(Node root, int target)
    if (root == null) return 0
    Map<TreeNode, TreeNode> parentMap = new HashMap<>()
    TreeNode targetNode = buildParentMap(root, target, parentMap)
    if (targetNode == null) return 0
    Queue<TreeNode> q = new LinkedList<>()
    Set<TreeNode> visited = new HashSet<>()
    q.offer(targetNode)
    visited.add(targetNode)
    int time = 0
    while (!q.isEmpty()) {
        int size = q.size()
        boolean anyBurnedThisSecond = false
        for (int i = 0; i < size; i++)
            TreeNode cur = q.poll()
            if (cur.left != null && !visited.contains(cur.left))
                visited.add(cur.left)
                q.offer(cur.left)
                anyBurnedThisSecond = true
            if (cur.right != null && !visited.contains(cur.right))
                visited.add(cur.right)
                q.offer(cur.right)
                anyBurnedThisSecond = true
            TreeNode parent = parentMap.get(cur)
            if (parent != null && !visited.contains(parent))
                visited.add(parent)
                q.offer(parent)
                anyBurnedThisSecond = true
        if (anyBurnedThisSecond) time++
    return time

void VERTICAL-VIEW(Node root)
    Map<Integer, List<Integer>> map;
    Queue<Pair> q;
    q.add(new Pair<>(root, 0));
    int min=0, max=0;
    while(!q.isEmpty())
        for(int i=0; i < q.size(); i++)
            Pair<Node, Integer> pair = q.poll();
            Node node  = pair.getKey();
            Integer dist = pair.getValue();
            if(map.get(dist) == null)
                map.put(dist, new ArrayList<>());
            map.get(dist).add(node.data);
            if(node.left != null)
                q.add(new Pair(node.left, dist - 1));
                min = Math.min(min, dist - 1);
            if(node.right != null)
                q.add(new Pair(node.right, dist + 1));
                max = Math.max(max, dist + 1);
        for(int i=min; i <= max; i++)
            println(map.get(i));

String SERIALIZE(Node root)
    StringBuilder sb = new StringBuilder();
    Queue<Node> q = new LinkedList<>();
    q.offer(root);
    while (!q.isEmpty())
         for (int i = 0; i < q.size(); i++)
            Node t = q.poll();
            if (t == null)
                sb.append("null ");
                    continue;
            sb.append(t.val).append(" ");
            q.offer(t.left);
            q.offer(t.right);
    return sb.toString();

            parent.right = right;
            q.offer(right);
     return root;
     

Node NEXT-RIGHT-POINTER(Node root) 
    Node curr = root;
    while (curr != null) 
    	Node next = curr.left;
    	if (next == null)  break;
    	while (curr != null) 
    		curr.left.next = curr.right;
    		if (curr.next != null) 
    			curr.right.next = curr.next.left;
    		curr = curr.next;
    	curr = next;
    return root;

int HOUSE-ROBBER-3(Node node, Map<Node, Integer> memo)
    if(node == null) return 0;
    if (memo.containsKey(node)) return memo.get(node);
    int sum = 0;
    if(node.left != null)
        sum += robHelper(node.left.left, memo) + robHelper(node.left.right, memo);
    if(node.right != null)
        sum += robHelper(node.right.left, memo) + robHelper(node.right.right, memo);
    sum = Math.max(sum+ node.val, robHelper(node.left, memo) + robHelper(node.right, memo));
    memo.put(node, sum);
    return sum;
