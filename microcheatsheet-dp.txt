int MIN-JUMPS(int[] nums)
    int jumps, max, curr = 0;
    for (int i = 0; i < nums.length; i++)
        if (curr < i)
            curr = max; jumps++;
        max = Math.max(max, i + nums[i]);

int FIND-MIN-COINS(int[] arr, int i, int target)
    if (target == 0) return 0;
    if (i < 0 || target < 0) return (int) Math.pow  
    int exclude = findMinCoins(arr, i - 1, target);
    int include = 1 + findMinCoins(arr, i, target - arr[i]);
    return Math.min(exclude, include);

int DECODE-WAYS(String s, int i)
    if (i == s.length()) return 1;
    if (s.charAt(i) == '0') return 0;   
    int count = dfs(s, i + 1);   
    if (i + 1 < s.length()) 
        int num = Integer.parseInt(s.substring(i, i + 2));
        if (num <= 26) count += dfs(s, i + 2);

int ROD-CUT(int[] price, int n)
   if (n == 0) return 0;
   int max = Integer.MIN_VALUE;
   for (int i = 1; i <= n; i++)
       int cost = price[i - 1] + rodCut(price, n - i);
       if (cost > max) max = cost;
    return max;
    
int BUY-SELL(int[] prices, int buySell, int ind)
    if (ind >= prices.length)  return 0;
    int profit = 0;
    if (buySell == 0) profit = Math.max(dfs(prices, 0, ind + 1),
                -prices[ind] + dfs(prices, 1, ind + 1));
    if (buySell == 1) profit = Math.max(dfs(prices, 1, ind + 1),
                prices[ind] + dfs(prices, 0, ind + 1));
    return profit;

int BUY-SELL-STOCK(int[] arr, int n, int ind, int buy, int cap, Map<String, Integer> map)
    if (ind >= n || cap == 0) return 0; // base case
    String key = ind + "|" + buy + "|" + cap;
    if (map.containsKey(key)) return map.get(key);
    int profit = 0;
    if (buy == 0)// We can buy the stock
    	profit = Math.max(0 + helper(arr, n, ind + 1, 0, cap, map),
    			-arr[ind] + helper(arr, n, ind + 1, 1, cap, map));
    if (buy == 1)// We can sell the stock
    	profit = Math.max(0 + helper(arr, n, ind + 1, 1, cap, map),
    			arr[ind] + helper(arr, n, ind + 1, 0, cap - 1, map));
    map.put(key, profit);
    return profit;
    
long BALL-ARRANGEMENT(int r, int g, int b, char prev)
    if (r == 0 && g == 0 && b == 0) return 1;
    String key = r + "," + g + "," + b + "," + prev;
    if (memo.containsKey(key)) return memo.get(key);
    long ways = 0;
    if (r > 0 && prev != 'R')
        ways += countWays(r - 1, g, b, 'R');
    if (g > 0 && prev != 'G')
        ways += countWays(r, g - 1, b, 'G');
    if (b > 0 && prev != 'B')
        ways += countWays(r, g, b - 1, 'B');
    memo.put(key, ways);
    return ways;

int DISTINCT-SUBSEQUENCE(String s, String t, int sIdx, int tIdx)
    if (memo[sIdx][tIdx] != -1) return memo[sIdx][tIdx];
    if (tIdx >= t.length()) return 1;
    if (sIdx >= s.length()) return 0;
    if (t.charAt(tIdx) == s.charAt(sIdx))
    	memo[sIdx][tIdx] =
    	    recursion(s, t, sIdx + 1, tIdx + 1, memo)
    	    + recursion(s, t, sIdx + 1, tIdx, memo);
    	return memo[sIdx][tIdx];
    memo[sIdx][tIdx] = recursion(s, t, sIdx + 1, tIdx, memo);
    return memo[sIdx][tIdx];
    
boolean INTERLEAVING(String A, String B, String C,
int i, int j, int k)
    if (k == C.length())
        return i == A.length() && j == B.length();
    String key = i + "," + j;
    if (memo.containsKey(key)) return memo.get(key);
    boolean valid = false;
    if (i < A.length() && A.charAt(i) == C.charAt(k))
        valid |= dfs(A, i+1, B, j, C, k+1, memo);
    if (j < B.length() && B.charAt(j) == C.charAt(k))
        valid |= dfs(A, i, B, j + 1, C, k + 1, memo);
    memo.put(key, valid);
    return valid;
    
List<Integer> LIS(int[] nums)
    int[] lengths, sequences = new int[n];
    Arrays.fill(lengths, 1);
    Arrays.fill(sequences, -1);
    int maxLen =1; int lastIndex = 0;
    for(int i=1; i < n; i++)
        for(int j=0; j < i; j++) 
            if(nums[i] >  nums[j] && 
            lengths[i] < lengths[j] + 1 ) 
                lengths[i] = lengths[j] + 1;
                sequences[i] = j;
        if (lengths[i] > maxLen) 
            maxLen = lengths[i];
            lastIndex = i;
    List<Integer> lis = new ArrayList<>();
    int index = lastIndex;
    while (index != -1) 
        lis.add(0, nums[index]);
        index = sequences[index];
    return lis;
    
String LCS(String s1, String s2)
    int[][] dp = new int[m + 1][n + 1];
    for (int i = 1; i <= m; i++)
        for (int j = 1; j <= n; j++)
            if (s1.charAt(i - 1) == s2.charAt(j - 1))
                dp[i][j] = dp[i - 1][j - 1] + 1;
             else
                dp[i][j] = Math.max(dp[i - 1][j], dp[i][j - 1]);
    StringBuilder lcs = new StringBuilder();
    int i = m, j = n;
    while (i > 0 && j > 0)
        if (s1.charAt(i - 1) == s2.charAt(j - 1))
            lcs.append(s1.charAt(i - 1)); i--; j--;
        else if (dp[i - 1][j] > dp[i][j - 1]) i--;
        else j--;
    return lcs.reverse().toString();
    
int LCSLength2(String X, String Y, int m, int n) 
    if (m == 0 || n == 0) return 0;
    if (X.charAt(m - 1) == Y.charAt(n - 1))
    	return 1 + LCSLength2(X, Y, m - 1, n - 1);
    return Integer.max(LCSLength2(X, Y, m - 1, n),
    LCSLength2(X, Y, m, n - 1));
    
List<String> WORD-BREAK(String s, Set<String> wordSet)
    if (memo.containsKey(s)) return memo.get(s);
    List<String> result = new ArrayList<>();
    if (s.length() == 0)
        result.add("");
        return result;
    for (int end = 1; end <= s.length(); end++)
        String prefix = s.substring(0, end);
        if (wordSet.contains(prefix))
            List<String> suffixBreaks = dfs(s.substring(end), wordSet, memo);
            for (String suffix : suffixBreaks)
                String space = suffix.isEmpty() ? "" : " ";
                result.add(prefix + space + suffix);
    memo.put(s, result);
    return result;

int OPTIMAL-PATH(int[][] grid)
    int[][] ways = new int[grid.length][grid[0].length];
    int rowStart = grid.length-1;
    for(int i= rowStart; i >= 0; i--)
    	for(int j= 0; j < grid[0].length; j++)
    		if(j == 0 && i== rowStart)
    			ways[i][j] = grid[i][j];
    		 else if(j== 0)
    			ways[i][j] = grid[i][j] + ways[i+1][j] ;
    		 else if (i == rowStart)
    			ways[i][j] = grid[i][j] + ways[i][j-1];
    		 else
    			ways[i][j] = grid[i][j]+Math.max(ways[i+1][j],ways[i][j-1]);
    return ways[0][grid[0].length-1];

int BURST-BALLOONS(int[] nums) 
    int[] newNums = new int[n+2];
    newNums[0] =1; newNums[n+1] = 1;
    for(int i=0; i < n; i++)
        newNums[i + 1] = nums[i];
    return helper(newNums, 0, n+1);
int helper  (int[] nums, int left, int right)
    if(left + 1 == right)  return 0;
    int maxCoins = 0;
    for(int i=left+1; i < right; i++)
        int coins = nums[left] * nums[i] * nums[right]
            + helper(nums, left, i)
            + helper(nums, i, right);
        maxCoins = Math.max(coins, maxCoins);
    return maxCoins;

int ACTIVITY-SELECTION(Activity[] activities, int i, Integer[] memo)
    if (i >= activities.length) return 0;
    if (memo[i] != null)  return memo[i];
    int maxActivities = recursiveSelect(activities, i + 1, memo);
    for (int j = i + 1; j < activities.length; j++)
        if (activities[i].finish <= activities[j].start)
            maxActivities = Math.max(maxActivities,
                1 + recursiveSelect(activities, j, memo));
    memo[i] = maxActivities;
    return maxActivities;
    
int JOB-SCHEDULING(int[] start, int[] end, int[] profit)
    int n = start.length;
    Job[] jobs = new Job[n];
    for (int i = 0; i < n; i++)
        jobs[i] = new Job(start[i], end[i], profit[i]);
    Arrays.sort(jobs, Comparator.comparingInt(j -> j.end));
    Map<Integer, Job> selectedJobMap = new HashMap<>();
    TreeMap<Integer, Integer> dp = new TreeMap<>();
    dp.put(0, 0);
    for (Job job : jobs)
        int prevProfit = dp.floorEntry(job.start).getValue();
        int currentProfit = prevProfit + job.profit;
        if (currentProfit > dp.lastEntry().getValue())
            dp.put(job.end, currentProfit);
            selectedJobMap.put(job.end, job);
    List<Job> selectedJobs = new ArrayList<>();
    Integer currentTime = dp.lastKey();
    while (currentTime != null && currentTime > 0)
        Job job = selectedJobMap.get(currentTime);
        if (job != null)
            selectedJobs.add(job);
            currentTime = dp.floorKey(job.start);
         else break;
    Collections.reverse(selectedJobs);
    return dp.lastEntry().getValue();

String SHORTEST-SUPERSTRING(String[] words)
    int n = words.length;
    int[][] overlap = new int[n][n];
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            if (i == j) continue;
            int max = Math.min(words[i].length(), words[j].length());
            for (int k = max; k >= 0; k--)
                if (words[i].endsWith(words[j].substring(0, k)))
                    overlap[i][j] = k;
                    break;
    int FULL = (1 << n) - 1;
    int[][] dp = new int[1 << n][n];
    int[][] parent = new int[1 << n][n];
    for (int mask = 0; mask <= FULL; mask++)
        Arrays.fill(dp[mask], Integer.MAX_VALUE / 2);
        Arrays.fill(parent[mask], -1);
    for (int i = 0; i < n; i++) dp[1 << i][i] = words[i].length();
    for (int mask = 1; mask <= FULL; mask++)
        for (int j = 0; j < n; j++)
            if ((mask & (1 << j)) == 0) continue;
            int prevMask = mask ^ (1 << j);
            if (prevMask == 0) continue;
            for (int k = 0; k < n; k++)
                if ((prevMask & (1 << k)) == 0) continue;
                int val = dp[prevMask][k] + words[j].length() - overlap[k][j];
                if (val < dp[mask][j])
                    dp[mask][j] = val;
                    parent[mask][j] = k;
    int minLen = Integer.MAX_VALUE, last = -1;
    for (int i = 0; i < n; i++)
        if (dp[FULL][i] < minLen)
            minLen = dp[FULL][i];
            last = i;
    StringBuilder sb = new StringBuilder();
    boolean[] used = new boolean[n];
    int mask = FULL;
    List<Integer> path = new ArrayList<>();
    while (last != -1)
        path.add(last);
        int temp = parent[mask][last];
        mask ^= (1 << last);
        last = temp;
    Collections.reverse(path);
    sb.append(words[path.get(0)]);
    for (int i = 1; i < path.size(); i++)
        int overlapLen = overlap[path.get(i-1)][path.get(i)];
        sb.append(words[path.get(i)].substring(overlapLen));
    return sb.toString();

int[] SMALLEST-SUFFICIENT-TEAM(String[] req_skills, List<List<String>> people) {
    int m = req_skills.length;
    int n = people.size();
    Map<String, Integer> skillIndex = new HashMap<>();
    for (int i = 0; i < m; i++)
        skillIndex.put(req_skills[i], i);
    int[] personMask = new int[n];
    for (int i = 0; i < n; i++)
        int mask = 0;
        for (String skill : people.get(i))
            if (skillIndex.containsKey(skill))
                mask |= 1 << skillIndex.get(skill);
        personMask[i] = mask;
    int FULL = (1 << m) - 1;
    List<Integer>[] dp = new List[1 << m];
    dp[0] = new ArrayList<>();
    for (int i = 0; i < n; i++)
        int pMask = personMask[i];
        if (pMask == 0) continue;
        List<Integer>[] dpCopy = dp.clone();
        for (int mask = 0; mask <= FULL; mask++)
            if (dpCopy[mask] == null) continue;
            int newMask = mask | pMask;
            if (dp[newMask] == null || dp[mask].size() + 1 < dp[newMask].size())
                List<Integer> newTeam = new ArrayList<>(dpCopy[mask]);
                newTeam.add(i);
                dp[newMask] = newTeam;
    List<Integer> result = dp[FULL];
    return result.stream().mapToInt(i -> i).toArray();

int NUMBER-WAYS-TO-WEAR-DIFF-HATS-TO-EACH-OTHER(List<List<Integer>> hats)
    int MOD = 1_000_000_007;
    int n = hats.size();
    List<Integer>[] hatToPeople = new List[41];
    for (int i = 1; i <= 40; i++) hatToPeople[i] = new ArrayList<>();
    for (int i = 0; i < n; i++)
        for (int h : hats.get(i))
            hatToPeople[h].add(i);
    int[][] dp = new int[41][1 << n];
    dp[0][0] = 1;
    for (int h = 1; h <= 40; h++)
        for (int mask = 0; mask < (1 << n); mask++)
            dp[h][mask] = dp[h - 1][mask];
            for (int p : hatToPeople[h])
                if ((mask & (1 << p)) != 0)
                    int prevMask = mask ^ (1 << p);
                    dp[h][mask] = (dp[h][mask] + dp[h - 1][prevMask]) % MOD;
    return dp[40][(1 << n) - 1];
