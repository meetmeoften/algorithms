    if (root == null || root == p || root == q) return root;
    Node left = lca(root.left, p, q);

Node LCA-WITH-PARENT-NODE(Node a, Node b)
    int depthA = getDepth(a);
    int depthB = getDepth(b);
    if (depthA > depthB)
        a = lift(a, depthA - depthB);
    else
        b = lift(b, depthB - depthA);
    while (a != b)
        a = a.parent; b = b.parent;
    return a;
int getDepth(Node x)
    int depth = 0;
    while (x.parent != null)
        x = x.parent; depth++;
    return depth;
Node lift(Node x, int steps)
    while (steps-- > 0)
        x = x.parent;
    return x;

    void init(int n) {

        up[v][0] = p;
        for (int i = 1; i < LOG; i++)
            if (up[v][i - 1] == -1) up[v][i] = -1;
            else up[v][i] = up[up[v][i - 1]][i - 1];
        for (int to : tree[v])
            if (to != p)
                depth[to] = depth[v] + 1;
                dfs(to, v);
    int kthAncestor(int v, int k)

            if ((k & (1 << i)) != 0)

        if (depth[u] < depth[v])
            int temp = u;
            u = v;
            v = temp;

int FIND-CLOSEST-IN-BST(BST tree, int target)

int findClosestValueInBst(BST tree, int target, int closest)
    BST currentNode = tree;
    while (currentNode != null)
    	var value1 = Math.abs(target - closest);
    	var value2 = Math.abs(target - currentNode.value);
    	if (value1 > value2) closest = currentNode.value;
    	if (target < currentNode.value) currentNode = currentNode.left;
    	else if (target > currentNode.value) currentNode = currentNode.right;
    	else break;
    return closest;

void FLATTEN-BINARY-TREE-LINKEDLIST(TreeNode root)
    if (root == null) return;
    Stack<TreeNode> st = new Stack<>();
    st.push(root);
    while (!st.isEmpty())
    	TreeNode curr = st.pop();
    	if (curr.right != null) st.push(curr.right);
    	if (curr.left != null) st.push(curr.left);
    	if (!st.isEmpty()) curr.right = st.peek();
    	curr.left = null;
    return;

Node NEXT-RIGHT-POINTERS(Node root)
    if (root == null) return null;
    Node curr = root;
    while (curr != null)
    	Node next = curr.left;
    	if (next == null) break;
    	while (curr != null)
    		curr.left.next = curr.right;
    		if (curr.next != null)
    			curr.right.next = curr.next.left;
    		curr = curr.next;
    	curr = next;
    return root;

void helper(TreeNode node, int targetSum, List<Integer> tempList, List<List<Integer>> list)
    if (node == null) return;
    tempList.add(node.val);
    if (node.left == null && node.right == null && targetSum - node.val == 0)
    	list.add(new ArrayList<>(tempList));
    	return;
    if (node.left != null)
    	helper(node.left, targetSum - node.val, tempList, list);
    	tempList.remove(tempList.size() - 1);
    if (node.right != null)
    	helper(node.right, targetSum - node.val, tempList, list);
    	tempList.remove(tempList.size() - 1);

List<TreeNode> DELETE-NODES-FOREST(TreeNode root, int[] deleteNodes)
    if (root == null) return new ArrayList<>();
    Set<TreeNode> resSet = new HashSet<>();
    resSet.add(root);
    if (deleteNodes.length == 0) return new ArrayList<>(resSet);
    Set<Integer> toDeleteSet = new HashSet<>();
    for (int val : deleteNodes) toDeleteSet.add(val);
    Queue<TreeNode> q = new LinkedList<>();
    q.offer(root);
    while (!q.isEmpty()) {
    	TreeNode node = q.poll();
    	if (toDeleteSet.contains(node.val)) {
    		resSet.remove(node);
    		if (node.left != null) resSet.add(node.left);
    		if (node.right != null) resSet.add(node.right);
    	if (node.left != null)
    		q.offer(node.left);
    		if (toDeleteSet.contains(node.left.val)) node.left = null;
    	if (node.right != null) {
    		q.offer(node.right);
    		if (toDeleteSet.contains(node.right.val)) node.right = null;
    return new ArrayList<>(resSet);

int OPTIMAL-COST-FOR-BST(int[] freq, int start, int end, int level) {
    if (start > end) return 0;
    int optimalCost = Integer.MAX_VALUE;
    for (int k = start; k <= end; k++)
        int left = findOptimalCost(freq, start, k - 1, level + 1);
        int right = findOptimalCost(freq, k + 1, end, level + 1);
        int cost = freq[k] * level + left + right;
        optimalCost = Integer.min(optimalCost, cost);
    return optimalCost;
